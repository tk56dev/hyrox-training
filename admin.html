初回セットアップ
データの追加・削除に Google Apps Script が必要です。一度だけ設定してください（約5分）。
STEP 1 — Google スプレッドシートを開く
トレーニングデータを保存しているスプレッドシートを開き、「拡張機能」→「Apps Script」をクリック
STEP 2 — 以下のコードをすべて貼り付けて保存
既存のコードを全削除し、以下をコピー＆ペーストして Ctrl+S で保存
function doGet(e) { const action = e.parameter.action; const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName('Sheet1') || ss.getSheets()[0]; const result = { success: false, action: action }; function fmtDate(v) { if (!v) return ''; if (typeof v === 'object') return Utilities.formatDate(v, 'Asia/Tokyo', 'yyyy-MM-dd'); return String(v).trim(); } function fmtTime(v) { if (!v) return ''; if (typeof v === 'object') return Utilities.formatDate(v, 'Asia/Tokyo', 'HH:mm'); return String(v).trim(); } function genId() { return 'hx_' + new Date().getTime() + '_' + Math.random().toString(36).slice(2, 6); } // ヘッダー自動初期化（列順: date,start,end,theme,type,notes,id,createdAt） const h = sheet.getRange(1,1,1,8).getValues()[0]; if (!h[0] || String(h[0]).toLowerCase() !== 'date') { sheet.getRange(1,1,1,8).setValues([ ['date','start','end','theme','type','notes','id','createdAt'] ]); } if (action === 'list') { const rows = sheet.getDataRange().getValues(); const entries = []; for (let i = 1; i < rows.length; i++) { const r = rows[i]; const d = fmtDate(r[0]); if (!d) continue; let id = String(r[6] || '').trim(); if (!id) { id = genId(); sheet.getRange(i+1, 7).setValue(id); } entries.push({ id, date: d, start: fmtTime(r[1]), end: fmtTime(r[2]), theme: String(r[3]||''), type: String(r[4]||'HYROX'), notes: String(r[5]||''), createdAt: String(r[7]||'') }); } result.success = true; result.entries = entries; } else if (action === 'add') { const id = genId(); const now = new Date().toISOString(); sheet.appendRow([ e.parameter.date || '', e.parameter.start || '', e.parameter.end || '', e.parameter.theme || '', e.parameter.type || 'HYROX', e.parameter.notes || '', id, now ]); result.success = true; result.id = id; } else if (action === 'delete') { const tid = e.parameter.id || ''; const tdate = e.parameter.date || ''; const ttheme = e.parameter.theme || ''; const rows = sheet.getDataRange().getValues(); for (let i = rows.length - 1; i >= 1; i--) { const rowId = String(rows[i][6]||'').trim(); const rowDate = fmtDate(rows[i][0]); const rowTheme = String(rows[i][3]||'').trim(); const hit = (tid && rowId === tid) || (!tid && rowDate === tdate && rowTheme === ttheme); if (hit) { sheet.deleteRow(i+1); result.success = true; break; } } } const cb = e.parameter.callback; const body = cb ? cb+'('+JSON.stringify(result)+')' : JSON.stringify(result); const mime = cb ? ContentService.MimeType.JAVASCRIPT : ContentService.MimeType.JSON; return ContentService.createTextOutput(body).setMimeType(mime); }
STEP 3 — ウェブアプリとしてデプロイ
1. 右上「デプロイ」→「新しいデプロイ」
2. 種類：「ウェブアプリ」を選択
3. アクセスできるユーザー：「全員」に設定
4. 「デプロイ」→ 表示された URL をコピー
